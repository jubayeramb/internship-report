\section{Key Tasks}

\subsection{Task 1: ArrayQ Platform Launch}
\begin{itemize}
    \item \textbf{Scope:} Develop and launch the new B2B product "ArrayQ" for haulers.
    \item \textbf{Objective:} To provide a dedicated SaaS platform for haulers to manage inventory and dumpster tracking.
    \item \textbf{Key Actions:}
    \begin{itemize}
        \item Developed Landing Page and Support pages.
        \item Implemented multi-tenant branding (dynamic logos/themes) to separate WeCycle and ArrayQ identities.
        \item Configured domains and SSL for the new platform.
    \end{itemize}
\end{itemize}

\subsection{Task 2: AI Chatbot Integration}
\begin{itemize}
    \item \textbf{Scope:} Deploy and integrate an AI-powered chatbot.
    \item \textbf{Objective:} To automate customer support and lead generation.
    \item \textbf{Key Actions:}
    \begin{itemize}
        \item Provisioned DigitalOcean droplets and configured Docker containers.
        \item Set up Nginx reverse proxy and SSL certificates.
        \item Integrated the chat widget into both Staging and Production environments.
    \end{itemize}
\end{itemize}

\subsection{Task 3: Financial Infrastructure (Stripe Connect)}
\begin{itemize}
    \item \textbf{Scope:} Modernize the payment and payout system.
    \item \textbf{Objective:} To allow automated payouts to haulers and better invoice management.
    \item \textbf{Key Actions:}
    \begin{itemize}
        \item Implemented Stripe Connected Accounts.
        \item Refactored Invoice IDs to be human-readable.
        \item Added dynamic payment links and logo rendering on invoices.
    \end{itemize}
\end{itemize}
